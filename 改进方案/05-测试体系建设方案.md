## 二、测试技术选型

### 2.1 后端测试框架

**单元测试 + 集成测试**
- **框架**: Jest (NestJS 官方推荐)
- **断言**: Jest 内置
- **Mock**: jest.mock() + @nestjs/testing
- **覆盖率**: jest --coverage

**E2E 测试**
- **框架**: Jest + Supertest
- **数据库**: Docker 启动临时 MySQL
- **HTTP 客户端**: Supertest

### 2.2 前端测试框架

**单元测试**
- **框架**: Vitest (Vue 3 官方推荐)
- **断言**: Vitest 内置
- **组件测试**: @vue/test-utils

**E2E 测试**
- **框架**: Playwright (替代 Cypress)
- **浏览器**: Chromium + Firefox
- **视觉回归**: Playwright 内置截图对比

### 2.3 测试数据管理

**数据准备**
- **Fixtures**: JSON/YAML 格式的测试数据
- **Factory**: TypeORM fixtures 或 factory-bot
- **数据库重置**: 每个测试前自动重置

**测试隔离**
- **事务回滚**: 每个测试在事务中运行，完成后回滚
- **独立数据库**: 每个测试套件使用独立 DB

```typescript
// src/modules/auth/auth.service.spec.ts
import { Test, TestingModule } from '@nestjs/testing';
import { AuthService } from './auth.service';
import { Repository } from 'typeorm';
import { User } from '../user/user.entity';
import { getRepositoryToken } from '@nestjs/typeorm';

describe('AuthService', () => {
  let service: AuthService;
  let userRepository: Repository<User>;

  // Mock Repository
  const mockUserRepository = {
    findOne: jest.fn(),
    save: jest.fn(),
    create: jest.fn(),
  };

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        AuthService,
        {
          provide: getRepositoryToken(User),
          useValue: mockUserRepository,
        },
      ],
    }).compile();

    service = module.get<AuthService>(AuthService);
    userRepository = module.get<Repository<User>>(getRepositoryToken(User));
  });

  afterEach(() => {
    jest.clearAllMocks();
  });

  it('should be defined', () => {
    expect(service).toBeDefined();
  });

  describe('login', () => {
    it('should return access token on successful login', async () => {
      const mockUser = {
        id: 1,
        username: 'test',
        password: 'hashed_password',
      };

      mockUserRepository.findOne.mockResolvedValue(mockUser);

      const result = await service.login({
        username: 'test',
        password: 'password123',
      });

      expect(result).toHaveProperty('access_token');
      expect(mockUserRepository.findOne).toHaveBeenCalledWith({
        where: { username: 'test' },
      });
    });

    it('should throw UnauthorizedException on wrong password', async () => {
      const mockUser = {
        id: 1,
        username: 'test',
        password: 'hashed_password',
      };

      mockUserRepository.findOne.mockResolvedValue(mockUser);

      await expect(
        service.login({
          username: 'test',
          password: 'wrong_password',
        })
      ).rejects.toThrow('用户名或密码错误');
    });
  });
});
```

#### 2.2 集成测试示例

**认证流程集成测试**
```typescript
// test/integration/auth.integration.spec.ts
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication } from '@nestjs/common';
import * as request from 'supertest';
import { AppModule } from '../../src/app.module';
import { setupTestDatabase, teardownTestDatabase } from '../setup';

describe('Authentication Integration Tests', () => {
  let app: INestApplication;

  beforeAll(async () => {
    await setupTestDatabase();

    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();
    await app.init();
  });

  afterAll(async () => {
    await app.close();
    await teardownTestDatabase();
  });

  describe('POST /auth/login', () => {
    it('should login successfully with valid credentials', async () => {
      const response = await request(app.getHttpServer())
        .post('/auth/login')
        .send({
          username: 'admin',
          password: 'admin123',
        })
        .expect(200);

      expect(response.body).toHaveProperty('data');
      expect(response.body.data).toHaveProperty('access_token');
      expect(response.body.data).toHaveProperty('refresh_token');
    });

    it('should reject invalid credentials', async () => {
      await request(app.getHttpServer())
        .post('/auth/login')
        .send({
          username: 'admin',
          password: 'wrong_password',
        })
        .expect(401);
    });

    it('should throttle after 5 failed attempts', async () => {
      // 连续失败 5 次
      for (let i = 0; i < 5; i++) {
        await request(app.getHttpServer())
          .post('/auth/login')
          .send({
            username: 'throttle_test',
            password: 'wrong',
          })
          .expect(401);
      }

      // 第 6 次应该被节流
      const response = await request(app.getHttpServer())
        .post('/auth/login')
        .send({
          username: 'throttle_test',
          password: 'wrong',
        })
        .expect(429);

      expect(response.body.message).toContain('请稍后再试');
    });
  });

  describe('GET /user/info', () => {
    it('should return user info with valid token', async () => {
      // 先登录获取 token
      const loginResponse = await request(app.getHttpServer())
        .post('/auth/login')
        .send({
          username: 'admin',
          password: 'admin123',
        });

      const token = loginResponse.body.data.access_token;

      // 使用 token 获取用户信息
      const response = await request(app.getHttpServer())
        .get('/user/info')
        .set('Authorization', `Bearer ${token}`)
        .expect(200);

      expect(response.body.data).toHaveProperty('username', 'admin');
      expect(response.body.data).toHaveProperty('roles');
    });

    it('should reject request without token', async () => {
      await request(app.getHttpServer())
        .get('/user/info')
        .expect(401);
    });
  });
});
```

```typescript
// test/e2e/declaration-flow.e2e-spec.ts
describe('Declaration Complete Flow (E2E)', () => {
  let app: INestApplication;
  let userToken: string;
  let enterpriseId: number;

  beforeAll(async () => {
    // 启动应用
    // 准备测试数据
  });

  it('should complete full declaration process', async () => {
    // Step 1: 登录
    const loginRes = await request(app.getHttpServer())
      .post('/auth/login')
      .send({ username: 'test', password: 'test123' })
      .expect(200);

    userToken = loginRes.body.data.access_token;

    // Step 2: 完善企业资料
    await request(app.getHttpServer())
      .post('/client/enterprise/profile')
      .set('Authorization', `Bearer ${userToken}`)
      .send({
        companyName: '测试企业',
        legalPerson: '张三',
        // ...完整资料
      })
      .expect(200);

    // Step 3: 选择政策项目
    const projectRes = await request(app.getHttpServer())
      .get('/policy/projects')
      .set('Authorization', `Bearer ${userToken}`)
      .expect(200);

    const projectId = projectRes.body.data[0].id;

    // Step 4: 创建申报
    const declarationRes = await request(app.getHttpServer())
      .post('/client/declarations')
      .set('Authorization', `Bearer ${userToken}`)
      .send({
        projectId,
        declarationYear: 2026,
      })
      .expect(200);

    const declarationId = declarationRes.body.data.id;

    // Step 5: 上传材料
    await request(app.getHttpServer())
      .post(`/client/declarations/${declarationId}/materials`)
      .set('Authorization', `Bearer ${userToken}`)
      .attach('file', './test/fixtures/sample.pdf')
      .expect(200);

    // Step 6: 提交申报
    await request(app.getHttpServer())
      .post(`/client/declarations/${declarationId}/submit`)
      .set('Authorization', `Bearer ${userToken}`)
      .expect(200);

    // Step 7: 验证申报状态
    const statusRes = await request(app.getHttpServer())
      .get(`/client/declarations/${declarationId}`)
      .set('Authorization', `Bearer ${userToken}`)
      .expect(200);

    expect(statusRes.body.data.status).toBe('submitted');
  });
});
```
